package main

import (
	"crypto/rand"
	"encoding/json"
	"fmt"
	"math/big"
	"os"
	"path/filepath"
	"sync"
	"time"

	"github.com/pquerna/otp"
	"github.com/pquerna/otp/totp"
	log "github.com/sirupsen/logrus"
	"golang.org/x/crypto/bcrypt"
)

const (
	mfaTokenTTL  = 5 * time.Minute
	totpIssuer   = "ovpn-admin"
	backupCodeLen = 8 // XXXX-XXXX format
)

// ── MFA Record ──────────────────────────────────────────────────────────────

type mfaRecord struct {
	Secret      string   `json:"secret"`       // base32-encoded TOTP secret
	Enabled     bool     `json:"enabled"`       // true after confirm
	BackupCodes []string `json:"backup_codes"`  // bcrypt hashes
	CreatedAt   string   `json:"created_at"`
}

// ── MFA Store (JSON file) ───────────────────────────────────────────────────

type mfaStore struct {
	mu   sync.RWMutex
	path string
	data map[string]mfaRecord // username → record
}

func newMfaStore(path string) *mfaStore {
	s := &mfaStore{path: path, data: make(map[string]mfaRecord)}
	s.load()
	return s
}

func (s *mfaStore) load() {
	data, err := os.ReadFile(s.path)
	if err != nil {
		return
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	_ = json.Unmarshal(data, &s.data)
}

func (s *mfaStore) save() {
	s.mu.RLock()
	data, _ := json.MarshalIndent(s.data, "", "  ")
	s.mu.RUnlock()
	if err := os.MkdirAll(filepath.Dir(s.path), 0755); err != nil {
		log.Warnf("mfa: cannot create dir for %s: %v", s.path, err)
		return
	}
	if err := os.WriteFile(s.path, data, 0600); err != nil {
		log.Warnf("mfa: cannot save %s: %v", s.path, err)
	}
}

func (s *mfaStore) get(username string) (mfaRecord, bool) {
	s.mu.RLock()
	defer s.mu.RUnlock()
	r, ok := s.data[username]
	return r, ok
}

func (s *mfaStore) set(username string, rec mfaRecord) {
	s.mu.Lock()
	s.data[username] = rec
	s.mu.Unlock()
	s.save()
}

func (s *mfaStore) delete(username string) {
	s.mu.Lock()
	delete(s.data, username)
	s.mu.Unlock()
	s.save()
}

func (s *mfaStore) isEnabled(username string) bool {
	r, ok := s.get(username)
	return ok && r.Enabled
}

// ── TOTP Verification ───────────────────────────────────────────────────────

func generateTOTPKey(username string) (*otp.Key, error) {
	return totp.Generate(totp.GenerateOpts{
		Issuer:      totpIssuer,
		AccountName: username,
		Period:      30,
		Digits:      otp.DigitsSix,
		Algorithm:   otp.AlgorithmSHA1,
	})
}

func verifyTOTPCode(secret, code string) bool {
	return totp.Validate(code, secret)
}

// ── Backup Codes ────────────────────────────────────────────────────────────

const backupCodeChars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789" // no 0/O/1/I ambiguity

func generateBackupCodes(count int) (plainCodes []string, hashedCodes []string) {
	plainCodes = make([]string, count)
	hashedCodes = make([]string, count)
	for i := 0; i < count; i++ {
		code := randomString(4, backupCodeChars) + "-" + randomString(4, backupCodeChars)
		plainCodes[i] = code
		hash, _ := bcrypt.GenerateFromPassword([]byte(code), bcrypt.DefaultCost)
		hashedCodes[i] = string(hash)
	}
	return
}

func randomString(length int, charset string) string {
	buf := make([]byte, length)
	for i := range buf {
		n, _ := rand.Int(rand.Reader, big.NewInt(int64(len(charset))))
		buf[i] = charset[n.Int64()]
	}
	return string(buf)
}

func verifyBackupCode(code string, hashes []string) bool {
	for _, h := range hashes {
		if bcrypt.CompareHashAndPassword([]byte(h), []byte(code)) == nil {
			return true
		}
	}
	return false
}

func consumeBackupCode(code string, hashes []string) []string {
	remaining := make([]string, 0, len(hashes))
	consumed := false
	for _, h := range hashes {
		if !consumed && bcrypt.CompareHashAndPassword([]byte(h), []byte(code)) == nil {
			consumed = true
			continue // skip this one
		}
		remaining = append(remaining, h)
	}
	return remaining
}

// ── MFA Token (intermediate, not a session) ─────────────────────────────────

type mfaTokenPayload struct {
	User    string `json:"u"`
	Exp     int64  `json:"exp"`
	Purpose string `json:"p"`
}

func signMfaToken(user string) string {
	return signMfaTokenWithTTL(user, mfaTokenTTL)
}

func signMfaTokenWithTTL(user string, ttl time.Duration) string {
	secret := sessionSecret()
	p := mfaTokenPayload{User: user, Exp: time.Now().Add(ttl).Unix(), Purpose: "mfa"}
	data, _ := json.Marshal(p)
	enc := base64.RawURLEncoding.EncodeToString(data)
	mac := computeHMAC(enc, secret)
	return enc + "." + mac
}

func verifyMfaToken(token string) (string, bool) {
	secret := sessionSecret()
	parts := strings.SplitN(token, ".", 2)
	if len(parts) != 2 {
		return "", false
	}
	enc, mac := parts[0], parts[1]
	if !hmac.Equal([]byte(computeHMAC(enc, secret)), []byte(mac)) {
		return "", false
	}
	raw, err := base64.RawURLEncoding.DecodeString(enc)
	if err != nil {
		return "", false
	}
	var p mfaTokenPayload
	if err := json.Unmarshal(raw, &p); err != nil {
		return "", false
	}
	if p.Purpose != "mfa" {
		return "", false
	}
	if time.Now().Unix() > p.Exp {
		return "", false
	}
	return p.User, true
}

package main

import (
	"encoding/json"
	"fmt"
	"net/http"
	"strings"

	log "github.com/sirupsen/logrus"
)

// mfaStatusHandler GET /api/mfa/status
func (oAdmin *OvpnAdmin) mfaStatusHandler(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
		return
	}
	user := oAdmin.sessionUser(r)
	if user == "" {
		http.Error(w, `{"error":"unauthorized"}`, http.StatusUnauthorized)
		return
	}
	enabled := oAdmin.mfaStore != nil && oAdmin.mfaStore.isEnabled(user)
	w.Header().Set("Content-Type", "application/json")
	fmt.Fprintf(w, `{"enabled":%v}`, enabled)
}

// mfaSetupHandler POST /api/mfa/setup — generate TOTP secret + QR URL
func (oAdmin *OvpnAdmin) mfaSetupHandler(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
		return
	}
	user := oAdmin.sessionUser(r)
	if user == "" {
		http.Error(w, `{"error":"unauthorized"}`, http.StatusUnauthorized)
		return
	}
	if oAdmin.mfaStore == nil {
		http.Error(w, `{"error":"MFA not configured"}`, http.StatusServiceUnavailable)
		return
	}

	key, err := generateTOTPKey(user)
	if err != nil {
		log.Errorf("mfa: generate key for %s: %v", user, err)
		http.Error(w, `{"error":"internal error"}`, http.StatusInternalServerError)
		return
	}

	oAdmin.mfaStore.set(user, mfaRecord{
		Secret:    key.Secret(),
		Enabled:   false,
		CreatedAt: time.Now().UTC().Format(time.RFC3339),
	})

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(map[string]string{
		"secret": key.Secret(),
		"qr_url": key.URL(),
	})
}

// mfaConfirmHandler POST /api/mfa/confirm — verify code, enable TOTP, return backup codes
func (oAdmin *OvpnAdmin) mfaConfirmHandler(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
		return
	}
	user := oAdmin.sessionUser(r)
	if user == "" {
		http.Error(w, `{"error":"unauthorized"}`, http.StatusUnauthorized)
		return
	}

	var req struct {
		Code string `json:"code"`
	}
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, `{"error":"invalid request"}`, http.StatusBadRequest)
		return
	}

	rec, ok := oAdmin.mfaStore.get(user)
	if !ok {
		http.Error(w, `{"error":"call /api/mfa/setup first"}`, http.StatusBadRequest)
		return
	}

	if !verifyTOTPCode(rec.Secret, req.Code) {
		http.Error(w, `{"error":"invalid code"}`, http.StatusUnauthorized)
		return
	}

	plainCodes, hashedCodes := generateBackupCodes(8)
	rec.Enabled = true
	rec.BackupCodes = hashedCodes
	oAdmin.mfaStore.set(user, rec)

	log.Infof("mfa: TOTP enabled for user %s", user)

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(map[string]interface{}{
		"backup_codes": plainCodes,
	})
}

// mfaDisableHandler DELETE /api/mfa — disable TOTP (requires valid code)
func (oAdmin *OvpnAdmin) mfaDisableHandler(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodDelete {
		http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
		return
	}
	user := oAdmin.sessionUser(r)
	if user == "" {
		http.Error(w, `{"error":"unauthorized"}`, http.StatusUnauthorized)
		return
	}

	var req struct {
		Code string `json:"code"`
	}
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, `{"error":"invalid request"}`, http.StatusBadRequest)
		return
	}

	rec, ok := oAdmin.mfaStore.get(user)
	if !ok {
		http.Error(w, `{"error":"MFA not configured"}`, http.StatusBadRequest)
		return
	}

	if !verifyTOTPCode(rec.Secret, req.Code) && !verifyBackupCode(req.Code, rec.BackupCodes) {
		http.Error(w, `{"error":"invalid code"}`, http.StatusUnauthorized)
		return
	}

	oAdmin.mfaStore.delete(user)
	log.Infof("mfa: TOTP disabled for user %s", user)

	w.Header().Set("Content-Type", "application/json")
	fmt.Fprint(w, `{"ok":true}`)
}

// mfaLoginHandler POST /api/login/mfa — second step of two-step login
func (oAdmin *OvpnAdmin) mfaLoginHandler(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
		return
	}

	ip := strings.Split(r.RemoteAddr, ":")[0]
	if !checkLoginRateLimit(ip) {
		http.Error(w, `{"error":"too many attempts"}`, http.StatusTooManyRequests)
		return
	}

	var req struct {
		MfaToken string `json:"mfa_token"`
		Code     string `json:"code"`
	}
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, `{"error":"invalid request"}`, http.StatusBadRequest)
		return
	}

	user, ok := verifyMfaToken(req.MfaToken)
	if !ok {
		recordLoginFailure(ip)
		http.Error(w, `{"error":"invalid or expired MFA token"}`, http.StatusUnauthorized)
		return
	}

	rec, hasRec := oAdmin.mfaStore.get(user)
	if !hasRec {
		http.Error(w, `{"error":"MFA not configured"}`, http.StatusUnauthorized)
		return
	}

	// Try TOTP code first, then backup code
	codeValid := verifyTOTPCode(rec.Secret, req.Code)
	if !codeValid {
		// Try backup code
		if verifyBackupCode(req.Code, rec.BackupCodes) {
			codeValid = true
			// Consume the backup code
			rec.BackupCodes = consumeBackupCode(req.Code, rec.BackupCodes)
			oAdmin.mfaStore.set(user, rec)
			log.Infof("mfa: backup code used for user %s (%d remaining)", user, len(rec.BackupCodes))
		}
	}

	if !codeValid {
		recordLoginFailure(ip)
		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(http.StatusUnauthorized)
		fmt.Fprint(w, `{"error":"invalid code"}`)
		return
	}

	recordLoginSuccess(ip)
	token := signSession(user)
	http.SetCookie(w, &http.Cookie{
		Name:     sessionCookieName,
		Value:    token,
		Path:     "/",
		HttpOnly: true,
		Secure:   true,
		SameSite: http.SameSiteStrictMode,
		MaxAge:   int(sessionTTL.Seconds()),
	})
	w.Header().Set("Content-Type", "application/json")
	fmt.Fprintf(w, `{"ok":true,"user":"%s"}`, user)
}

// sessionUser extracts username from session cookie (helper for MFA handlers)
func (oAdmin *OvpnAdmin) sessionUser(r *http.Request) string {
	cookie, err := r.Cookie(sessionCookieName)
	if err != nil {
		return ""
	}
	user, ok := verifySession(cookie.Value)
	if !ok {
		return ""
	}
	return user
}

export async function loginMfa(mfaToken, code) {

async function startSetup() {
  loading.value = true
  error.value = ''
  try {
    const data = await setupMfa()
    secret.value = data.secret
    qrUrl.value = data.qr_url
    step.value = 'setup'
